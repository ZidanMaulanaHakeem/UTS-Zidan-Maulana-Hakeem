Skema PostgreSQL & Panduan Implementasi

1. Struktur Tabel (Soal 1 & Soal 6)

-- Chart of Accounts
CREATE TABLE chart_of_accounts (
    account_id SERIAL PRIMARY KEY,
    account_code VARCHAR(50) UNIQUE NOT NULL,
    account_name VARCHAR(150) NOT NULL,
    normal_balance VARCHAR(10) CHECK (normal_balance IN ('DEBIT', 'CREDIT')),
    report_class VARCHAR(50) NOT NULL
);

-- Journal Headers dengan Audit Trail & Checksum
CREATE TABLE journal_headers (
    header_id SERIAL PRIMARY KEY,
    transaction_date DATE NOT NULL,
    reference_no VARCHAR(100) UNIQUE NOT NULL,
    description TEXT NOT NULL,
    posting_status VARCHAR(20) DEFAULT 'DRAFT' CHECK (posting_status IN ('DRAFT', 'POSTED', 'REVERSED')),
    created_by VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    posted_at TIMESTAMP NULL,
    hash_checksum VARCHAR(64) NOT NULL
);

-- Journal Lines
CREATE TABLE journal_lines (
    line_id SERIAL PRIMARY KEY,
    header_id INT REFERENCES journal_headers(header_id) ON DELETE CASCADE,
    account_id INT REFERENCES chart_of_accounts(account_id),
    debit NUMERIC(15,2) DEFAULT 0.00 CHECK (debit >= 0),
    credit NUMERIC(15,2) DEFAULT 0.00 CHECK (credit >= 0),
    CONSTRAINT chk_single_side CHECK ((debit > 0 AND credit = 0) OR (credit > 0 AND debit = 0))
);

2. Audit Trail & Pencegahan Ubah/Hapus Status POSTED

CREATE OR REPLACE FUNCTION prevent_posted_modification()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.posting_status = 'POSTED' THEN
        RAISE EXCEPTION 'Transaksi berstatus POSTED tidak dapat diubah atau dihapus (Audit Trail Protection).';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_posted_edit
BEFORE UPDATE OR DELETE ON journal_headers
FOR EACH ROW EXECUTE FUNCTION prevent_posted_modification();

3. Skema Persediaan & Trigger Stok Negatif (Soal 7)

CREATE TABLE inventory_items (
    item_id SERIAL PRIMARY KEY,
    item_code VARCHAR(50) UNIQUE NOT NULL,
    item_name VARCHAR(150) NOT NULL,
    stock_qty INT DEFAULT 0 CHECK (stock_qty >= 0)
);

CREATE TABLE inventory_transactions (
    inv_trx_id SERIAL PRIMARY KEY,
    item_id INT REFERENCES inventory_items(item_id),
    qty_change INT NOT NULL,
    trx_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE OR REPLACE FUNCTION check_negative_stock()
RETURNS TRIGGER AS $$
DECLARE
    current_stock INT;
BEGIN
    SELECT stock_qty INTO current_stock FROM inventory_items WHERE item_id = NEW.item_id;
    IF (current_stock + NEW.qty_change) < 0 THEN
        RAISE EXCEPTION 'Stok tidak mencukupi. Transaksi ditolak untuk mencegah stok negatif.';
    END IF;
    
    UPDATE inventory_items 
    SET stock_qty = stock_qty + NEW.qty_change 
    WHERE item_id = NEW.item_id;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_inventory_stock_control
AFTER INSERT ON inventory_transactions
FOR EACH ROW EXECUTE FUNCTION check_negative_stock();

Modul Persediaan & Validasi Stok

Modul persediaan dirancang dengan tabel terpisah (inventory_items dan inventory_transactions). Sesuai instruksi soal, trigger database aktif memeriksa ketersediaan stok setiap kali mutasi persediaan dicatat, menolak transaksi yang menyebabkan stok negatif secara atomik di level server.

Integrasi ke Jurnal Nominal baru akan berjalan setelah mapping akun persediaan dan harga pokok penjualan (HPP) didefinisikan secara resmi.
